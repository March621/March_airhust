bool mission_pos_cruise(float x, float y, float z, float target_yaw, float error_max)
{                        
    if (mission_pos_cruise_flag == false)
    {
        mission_pos_cruise_last_position_x = local_pos.pose.pose.position.x;
        mission_pos_cruise_last_position_y = local_pos.pose.pose.position.y;
        mission_pos_cruise_flag = true;
    }
    //修改掩码
    setpoint_raw.type_mask = 1 + 2 + 4 + 64 + 128 + 256 + 2048;
    setpoint_raw.coordinate_frame = 1;
    setpoint_raw.position.x = x + init_position_x_take_off;
    setpoint_raw.position.y = y + init_position_y_take_off;
    setpoint_raw.position.z = z + init_position_z_take_off;
    setpoint_raw.yaw = 0;
    ROS_INFO("now (%.2f,%.2f,%.2f,%.2f) to ( %.2f, %.2f, %.2f, %.2f), init_pose(%.2f, %.2f, %.2f, %.2f)", 
        local_pos.pose.pose.position.x ,local_pos.pose.pose.position.y, local_pos.pose.pose.position.z, yaw * 180.0 / M_PI, 
        x, y, z, target_yaw * 180.0 / M_PI, 
        init_position_x_take_off, init_position_y_take_off, init_position_z_take_off, init_yaw_take_off * 180.0 / M_PI);
    double error_x, error_y, error_z;
    error_x = x + init_position_x_take_off - local_pos.pose.pose.position.x;
    error_y = y + init_position_y_take_off - local_pos.pose.pose.position.y;
    error_z = z + init_position_z_take_off - local_pos.pose.pose.position.z;
    if (fabs(error_x) < error_max && fabs(error_y) < error_max && fabs(error_z) < error_max && fabs(yaw - target_yaw) < 0.1)
    {
        ROS_INFO("到达目标点，巡航点任务完成");
        mission_pos_cruise_flag = false;
        //到达目标点之后速度归零
        setpoint_raw.velocity.x = 0;
        setpoint_raw.velocity.y = 0;
        return true;
    }
    else
    {
        //计算速度,k取0.5
        setpoint_raw.velocity.x = error_x * 0.5;
        setpoint_raw.velocity.y = error_y * 0.5;
        //v_max没有赋值
        double v_x, v_y, v ,v_max;
        v_x = setpoint_raw.velocity.x;
        v_y = setpoint_raw.velocity.y;
        v = sqrt(v_x * v_x + v_y * v_y);
        setpoint_raw.velocity.z = 0;
        //合速度限速
        if( v > v_max)
        {
            setpoint_raw.velocity.x = v_max*v_x/v;
            setpoint_raw.velocity.y = v_max*v_y/v;
        }
        return false;
    }
}
